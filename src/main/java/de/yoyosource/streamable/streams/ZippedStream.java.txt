package de.yoyosource.streamable.streams;

import de.yoyosource.streamable.Streamable;
import de.yoyosource.streamable.StreamableGatherer;

import java.util.function.BiFunction;
import java.util.function.BiPredicate;
import java.util.function.Consumer;

public interface ZippedStream<A, B> extends Streamable<ZippedStream.Zip<A, B>> {

    static <A, B> Class<ZippedStream<A, B>> type() {
        return (Class<ZippedStream<A, B>>) (Class) ZippedStream.class;
    }

    class Zip<A, B> {
        public final A a;
        public final B b;

        public Zip(A a, B b) {
            this.a = a;
            this.b = b;
        }
    }

    default <C> Streamable<C> map(BiFunction<A, B, C> mapper) {
        return gather(new StreamableGatherer<>() {
            @Override
            public boolean apply(Zip<A, B> input, Consumer<C> next) {
                next.accept(mapper.apply(input.a, input.b));
                return false;
            }

            @Override
            public void finish(Consumer<C> next) {
            }
        });
    }

    default ZippedStream<A, B> filter(BiPredicate<A, B> predicate) {
        return gather(new StreamableGatherer<>() {
            @Override
            public boolean apply(Zip<A, B> input, Consumer<Zip<A, B>> next) {
                if (predicate.test(input.a, input.b)) {
                    next.accept(input);
                }
                return false;
            }

            @Override
            public void finish(Consumer<Zip<A, B>> next) {

            }
        });
    }

    default Streamable<A> getLeft() {
        return gather(new StreamableGatherer<>() {
            @Override
            public boolean apply(Zip<A, B> input, Consumer<A> next) {
                next.accept(input.a);
                return false;
            }

            @Override
            public void finish(Consumer<A> next) {

            }
        });
    }

    default Streamable<B> getRight() {
        return gather(new StreamableGatherer<>() {
            @Override
            public boolean apply(Zip<A, B> input, Consumer<B> next) {
                next.accept(input.b);
                return false;
            }

            @Override
            public void finish(Consumer<B> next) {

            }
        });
    }
}
